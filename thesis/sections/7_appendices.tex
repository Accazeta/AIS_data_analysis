\chapter[Pseudocodice della pipeline]{\rev{Pseudocodice della pipeline}}\label{app:pseudocode}

\begin{revblock}
Gli Algoritmi~\ref{alg:app_discovery}, \ref{alg:app_model} e~\ref{alg:app_score} riassumono le tre
fasi della pipeline descritte nel Capitolo~\ref{chap:methodology}. Lo pseudocodice segue
l'implementazione e non l'algoritmo originale di Pallotta et al.~\cite{pallotta}. Per brevità sono
omessi i passaggi di importazione e pulizia descritti nella Sezione~\ref{sec:meth_preprocessing},
e $\mathrm{GCD}(a, b)$ indica la Great-Circle Distance tra due posizioni $a$ e $b$.
\end{revblock}

\begin{algorithm}[htbp]
\caption[Classificazione degli stati, estrazione dei waypoint e delle rotte.]{\rev{Classificazione degli stati, estrazione dei waypoint e delle rotte.}}\label{alg:app_discovery}
\begin{revblock}
\begin{algorithmic}[1]
\Require record AIS puliti, soglie $T_{\mathrm{lost}} = 620$~s, $T_{\mathrm{exit}} = 12$~h,
    $v_{\max} = 102$~nodi, $v_s = 0{,}5$~nodi
\Ensure waypoint, etichetta di rotta di ogni punto, trip
\ForAll{nave}
    \State ordina i punti della nave per \texttt{BaseDateTime}
    \ForAll{coppia di punti consecutivi $(i-1, i)$}
        \State $\Delta t \gets t_i - t_{i-1}$, \quad $v_i \gets \mathrm{GCD}(x_{i-1}, x_i) / \Delta t$
        \If{$\Delta t > T_{\mathrm{lost}}$}
            \State $v_i \gets \texttt{SOG}_i$ se $\texttt{SOG}_i < v_{\max}$
            \State stato di $i-1$ e $i$: \emph{lost} e \emph{return}, oppure \emph{exit} ed \emph{entry} se $\Delta t > T_{\mathrm{exit}}$
        \EndIf
    \EndFor
    \State elimina i punti con $v_i \geq v_{\max}$
    \State punti senza stato: \emph{stationary} se $v_i \leq v_s$, altrimenti \emph{sailing}
    \State primo punto \emph{entry}, ultimo punto \emph{exit}; elimina gli \emph{entry} e gli \emph{exit} consecutivi ripetuti
\EndFor
\State $P \gets$ punti \emph{stationary}, \emph{entry} ed \emph{exit}
\State aggrega $P$ per coppia (\texttt{LAT}, \texttt{LON}) con peso $w = \max(1, \mathrm{round}(\log_2 n))$
\State replica ogni coordinata $w$ volte ed esegui H-DBSCAN
\ForAll{cluster $k$}
    \State waypoint $\texttt{WP\_}k \gets$ inviluppo convesso dei punti del cluster
\EndFor
\ForAll{nave e periodo (\emph{entry}, \emph{exit}) della nave}
    \ForAll{punto $i$ del periodo}
        \If{$i$ è in un waypoint e ($v_i \leq v_s$ oppure $i$ è \emph{entry} o \emph{exit})}
            \State etichetta di $i \gets$ nome del waypoint significativo
        \Else
            \State etichetta di $i \gets$ \texttt{WP\_X - WP\_Y}, con \texttt{WP\_X} e \texttt{WP\_Y}
                waypoint significativi precedente e successivo (\texttt{<NA>} se assenti)
        \EndIf
    \EndFor
\EndFor
\State trip $\gets$ sequenze di punti consecutivi con la stessa etichetta di rotta \texttt{WP\_X - WP\_Y}
\end{algorithmic}
\end{revblock}
\end{algorithm}

\begin{algorithm}[htbp]
\caption[Stima del modello di normalità di una rotta $R$.]{\rev{Stima del modello di normalità di una rotta {$R$}.}}\label{alg:app_model}
\begin{revblock}
\begin{algorithmic}[1]
\Require punti e trip della rotta $R$, raggio dello stato $d = 500$~m, 11 lag
\Ensure densità $p_j$, termini $L_i$, parametri $m$ e $\beta$
\State $f_{\mathrm{geo}} \gets$ KDE su (\texttt{LON}, \texttt{LAT}) di tutti i punti di $R$
\State $f_{v} \gets$ KDE su (\texttt{LON}, \texttt{LAT}) con pesi pari a \texttt{SOG}
\ForAll{punto $j$ di $R$}
    \State $p_j \gets f_{\mathrm{geo}}(x_j) \cdot f_{v}(x_j)$
\EndFor
\ForAll{punto $i$ di $R$}
    \State $N_i \gets \{ j : \mathrm{GCD}(x_i, x_j) < d \}$, \quad $L_i \gets \sum_{j \in N_i} \log p_j$
\EndFor
\State $\tau_{\min} \gets \max(30~\mathrm{s}, D_{\min}/20)$, \quad $\tau_{\max} \gets \min(D_{\mathrm{med}}/5, 3.600~\mathrm{s})$
\State $\tau_1, \dots, \tau_{11} \gets$ lag equidistanti tra $\tau_{\min}$ e $\tau_{\max}$
\ForAll{trip di $R$ e punto $i$ del trip, escluso il primo}
    \ForAll{lag $\tau_k$}
        \State $j \gets$ punto del trip con timestamp più vicino a $t_i - \tau_k$
        \If{$|(t_i - t_j) - \tau_k| \leq \mathrm{toll}(\tau_k) \cdot \tau_k$}
            \State $\hat{x}_i \gets x_j + (V_{x,j}, V_{y,j}) \cdot (t_i - t_j)$, convertito in gradi
            \State aggiungi $\Delta p = \mathrm{GCD}(\hat{x}_i, x_i)$ all'insieme $D_k$
        \EndIf
    \EndFor
\EndFor
\ForAll{lag $\tau_k$}
    \State $(\alpha_k, \beta_k) \gets$ stima di massima verosimiglianza di una Weibull su $D_k$
\EndFor
\State $m \gets$ regressione ai minimi quadrati di $\alpha_k = m \cdot \tau_k$, \quad $\beta \gets$ media dei $\beta_k$
\end{algorithmic}
\end{revblock}
\end{algorithm}

\begin{algorithm}[htbp]
\caption[Punteggio di normalità dei trip e classificazione.]{\rev{Punteggio di normalità dei trip e classificazione.}}\label{alg:app_score}
\begin{revblock}
\begin{algorithmic}[1]
\Require trip della rotta $R$, termini $L_i$, parametri $m$ e $\beta$, probabilità a priori $P(R)$
\Ensure punteggio di ogni trip, ranking e trip anomali
\ForAll{trip $T = (x_1, \dots, x_n)$}
    \State $O(T) \gets \frac{1}{n} \sum_{i=1}^{n} L_i$ \Comment{termine di osservazione}
    \For{$t = 2, \dots, n$}
        \State $\Delta t_t \gets t_t - t_{t-1}$, \quad $\hat{x}_t \gets x_{t-1} + (V_{x,t-1}, V_{y,t-1}) \cdot \Delta t_t$
        \State $\Delta p_t \gets \mathrm{GCD}(\hat{x}_t, x_t)$
    \EndFor
    \State $Tr(T) \gets -\frac{1}{n-1} \sum_{t=2}^{n} {\left( \Delta p_t / (m \cdot \Delta t_t) \right)}^{\beta}$ \Comment{termine di transizione}
    \State $S(T) \gets O(T) + Tr(T) + \log P(R)$
\EndFor
\State ordina i trip per $S(T)$ decrescente
\State $s \gets Q_1 - 1{,}5 \cdot (Q_3 - Q_1)$, con $Q_1$ e $Q_3$ quartili dei punteggi
\State \Return trip con $S(T) < s$ come anomali
\end{algorithmic}
\end{revblock}
\end{algorithm}
